\begin{frame}{Estampilles de Lamport}

  \onBlock[top=-6mm]{Définition -- Estampille de Lamport}{
    L'\structure{estampille de Lamport} d'un pas $a$ est le couple \alert{$\langle \TS, i \rangle$} tel que :
    \begin{description}[xxx]
    \item[\alert{$\TS$}] : date de Lamport de $a$
    \item[\alert{$i$}] : identifiant du processus où se produit $a$
    \end{description}
  }

  \onBlock[bottom=-2mm]{Remarques -- Ordre lexicographique sur les estampilles}{
    \begin{itemize}
    \item\vspace{-1.5mm} On considère l'ordre :
      $
      \alert{\langle \TS_i, i \rangle <_{lex} \langle \TS_j, j \rangle
        \quad \Leftrightarrow \quad
        \TS_i < \TS_j \lor \left( \TS_i = \TS_j \land i<j \right)}
      $
    \item\vspace{-.5mm} La relation $<_{lex}$ décrit un \alert{ordre total} sur les pas estampillés
    \item\vspace{-1mm} Compatibilité avec \emph{happened before} : $\alert{\forall a, b,\quad a \hb b \quad\Rightarrow\quad a <_{lex} b}$
    \item\vspace{-.5mm} Il peut exister un pas d'estampille antérieure inconnu par un processus
    \end{itemize}
  }
