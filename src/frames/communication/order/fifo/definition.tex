\SetKwBroadcast{FIFO}{fifo}
\SetKwBroadcast{RBFIFO}{rb-fifo}
\SetKwBroadcast{URBFIFO}{urb-fifo}


\begin{frame}{Définition de FIFO Broadcast}

  \onBlock[top=-3mm]{Propriétés de \FIFO-broadcast}{
    \begin{description}
    \item[Validité :] Si $p_i$ délivre $m$ de $p_j$, $p_j$ a diffusé $m$
    \item[Intégrité :] Un processus $p_i$ délivre $m$ au plus une fois
    \item[Terminaison :] Tout appel de \Broadcast par un correct se termine
    \item[Complétion :] Si un correct diffuse $m$, tous les corrects délivrent $m$
    \item[\alert{Ordonnancement FIFO :}] Si un processus diffuse $m$ avant $m'$, \\alors aucun processus ne délivre $m'$ avant $m$
    \end{description}
  }

  \on[y=-12mm]{
    \begin{tikzpicture}[y=10mm]
      \draw[process] (0,1) node[left]{$p_1$} to (10,1);
      \draw[process] (0,0) node[left]{$p_2$} to (10,0);

      \coordinate                  (m1) at (1,1);
      \node[structure, above]   at (m1) {$m$} ;
      \draw[structure, message]    (m1.center) to[bend left] (2,1) ;
      \draw[structure, message]    (m1.center) to            (5,0) ;

      \coordinate                (m2) at (3,1);
      \node[example, above]   at (m2) {$m'$} ;
      \draw[example, message]    (m2.center) to[bend left] (4,1) ;
      \draw[example, message]    (m2.center) to            (4,0) ;

      \draw[alert, thick] (6,1.25) -- (9,-.25);
      \draw[alert, thick] (9,1.25) -- (6,-.25);
    \end{tikzpicture}
  }
  
  \onBlock[bottom]{Remarque}{
    On peut renforcer l'abstraction par des propriétés de vivacité :
    \begin{itemize}
    \item \RBFIFO-broadcast, \URBFIFO-broadcast
    \end{itemize}
  }
  
\end{frame}
