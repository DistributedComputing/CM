\SetKwBroadcast{CB}{cb}
\SetKwBroadcast{RBCB}{rb-cb}
\SetKwBroadcast{URBCB}{urb-cb}

\begin{frame}{Définition de Causal Broadcast}

  \onBlock[top=-5mm]{Propriétés de \CB-broadcast}{
    \begin{description}
    \item[Validité :]\vspace{-2mm} Si $p_i$ délivre $m$ de $p_j$, $p_j$ a diffusé $m$
    \item[Intégrité :] Un processus $p_i$ délivre $m$ au plus une fois
    \item[Terminaison :] Tout appel de \Broadcast par un correct se termine
    \item[Complétion :] Si un correct diffuse $m$, tous les corrects délivrent $m$
    \item[Ordonnancement FIFO :]
      Si un processus diffuse $m$ avant $m'$, \\
      alors aucun processus ne délivre $m'$ avant $m$
    \item[\alert{Ordonnancement causal :}]
      Si un processus délivre $m$ avant de diffuser $m'$, \\
      alors aucun processus ne délivre $m'$ avant $m$
    \end{description}
  }

  \onBlock[bottom=1mm]{Remarques}{
    \vspace{-1mm}
    \begin{itemize}
    \item\vspace{-1mm} Les livraisons respectent l'ordre \emph{happened before} des diffusions\\
      \begin{center}
        \footnotesize
        $\structure{\Broadcast(m)} \alert{~\hb~} \structure{\Broadcast(m')}
        \quad \Rightarrow \quad
        \structure{\Deliver~m} \text{ at } p_i \alert{~\hb~} \structure{\Deliver~m'} \text{ at } p_i$
      \end{center}
    \item \structure{Ajout de propriétés :} \RBCB-broadcast, \URBCB-broadcast, ...
    \end{itemize}
  }

  \on[y=-15mm, x=10mm]{
