\begin{frame}{Implémentation de Causal Broadcast}

  \on<-2>[top]{
    \begin{algorithm}[H]
      \LVariables{\At $p_i$}{
        \Example{$\Clock_i$}\uncover<2>{\tcp*[f]{$\Delivered_i[1..n]$}}
      }
      \Method{$\CB.\Broadcast(\Payload)$ \At $p_i$}{
       \nl \alertb<1>{\URB}.\SBroadcast $\alert{\MSG(\Payload, \Example{\Clock_i.\Now(i)})}$\uncover<2>{\tcp*[r]{$\Delivered_i$}}
      }
      \Upon{\alertb<1>{\URB}.\Deliver $\alert{\MSG(\Payload, \Date)}$ \From $p_i$ \At $p_j$}{
       \nl \Wait{\Example{$\Clock_j.\Now() \ge \Date$}}\uncover<2>{\tcp*[r]{$\forall k, \Delivered_j[k] \ge \Date[k]$}}
       \nl \Example{$\Clock_j.\Update(i)$}\uncover<2>{\tcp*[r]{$\Delivered_j[i] \leftarrow \Date[i]+1$}}
       \nl \Trigger \CB.\Deliver $\Payload$ \From $p_i$
      }
    \end{algorithm}
  }

  \ob<3->[top]{
    \begin{algorithm}[H]
      \LVariables{\At $p_i$}{
        $\Delivered_i \leftarrow [0, ..., 0]$\tcp*[f]{Compteurs des messages délivrés}
      }
      \Method{$\CB.\Broadcast(\Payload)$ \At $p_i$}{
        \nl \URB.\SBroadcast $\alert{\MSG(\Payload, \Delivered_i)}$
      }
      \Upon{\URB.\Deliver $\alert{\MSG(\Payload, \Date)}$ \From $p_i$ \At $p_j$}{
        \nl \Wait{$\forall k \in \{1, ..., n\}, \Delivered_j[k] \ge \Date[k]$}\\
        \nl $\Delivered_j[i] \leftarrow \Date[i] + 1$\\
        \nl \Trigger \CB.\Deliver $\Payload$ \From $p_i$
      }
    \end{algorithm}
  }

  \onBlock<2>[anchor=north, y=-2mm]{Notion d'horloge vectorielle}{
    \begin{description}[Horloge :]
    \item[Temps :]\vspace{-2mm} capture \alert{exactement} la relation happens-before\\[-3mm]
      $$\forall a, b, ~ a \hb b \Leftrightarrow \Time(a) < \Time(b)$$
    \item[Date :] \alert{vecteur de version} $\Date[1..n]$ formé d'un entier par processus
      \begin{itemize}
      \item $\Date[k]$ est le nombre de messages délivrés de $p_k$
      \item Comparaison : $\Date_1 \le \Date_2$ si $\forall k, \Date_1[k] \le \Date_2[k]$
      \end{itemize}
    \end{description}
  }
