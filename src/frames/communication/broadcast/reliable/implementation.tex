\begin{frame}{Implémentation de Uniform Reliable Broadcast}
  
  \on[top=-3mm]{
    \begin{algorithm}[H]
      \LVariables{\At $p_i$}{
        $\Next_i \leftarrow 0$\tcp*[r]{Compteur pour numéroter les messages de $p_i$}
        $\Delivered_i \leftarrow \emptyset$\tcp*[f]{Ensemble des identifiants de messages délivrés}
      }
      \Method{$\URB.\Broadcast(\Payload)$ \At $p_i$}{
        \nl \STA.\Broadcast $\alert{\MSG(\Payload, \Next_i, i)}$\\
        \nl $\Next_i \leftarrow \Next_i+1$
      }
      
      \Upon{\STA.\Deliver $\alert{\MSG(\Payload, \SN, i)}$ \From $p_k$ \At $p_j$}{
        \nl \If{$\langle \SN, i\rangle \notin \Delivered_j$}{
          \nl \Alert<4-5,9>{\STA.\Broadcast $\alert{\MSG(\Payload, \SN, i)}$}\\
          \nl \Trigger \Alert<4-5,9>{\URB.\Deliver $\Payload$ \From $p_i$}\\
          \nl $\Delivered_j \leftarrow \Delivered_j \cup \{\langle \SN, i\rangle\}$
        }
      }
    \end{algorithm}
  }

  \obBlock<4>[anchor=north, y=-9mm]{Démonstration : Validité}{
    \vspace{-2mm}
    \begin{center}
      \structure{Si $p_j$ délivre $m$ de $p_i$, $p_i$ a diffusé $m$}
    \end{center}
    \vspace{-2mm}
    \begin{description}[STA-Terminaison :]
    \item[Ligne 5 :] $p_j$ a délivré $\alert{\MSG(\Payload, \SN, i)}$ de $p_k$
    \item[Validité :] $p_k$ a diffusé $\alert{\MSG(\Payload, \SN, i)}$
    \item[HB bien-fondé :] il y a une première diffusion de $\alert{\MSG(\Payload, \SN, i)}$
    \item[Lignes 1 et 4 :] ce premier diffuseur est $p_i$
    \end{description}
  }

  \obBlock<5>[anchor=north, y=-9mm]{Démonstration : Intégrité}{
    \vspace{-2mm}
    \begin{center}
      \structure{Un processus $p_j$ délivre un message $m$ au plus une fois}
    \end{center}
    \vspace{-2mm}
    \begin{description}[STA-Terminaison :]
    \item[Invariant :] $\Delivered_j$ croissant selon $\subseteq$
    \item[Ligne 3 :] avant livraison,  $\langle \SN, i\rangle\notin\Delivered_j$
    \item[Ligne 6 :] après livraison, $\langle \SN, i\rangle\in\Delivered_j$
    \end{description}
  }

  \obBlock<7>[anchor=north, y=-9mm]{Démonstration : Complétion}{

    \end{center}
    \vspace{-2mm}
    \begin{description}[STA-Terminaison :]

    \item[Fiabilité :] $p_k$ délivre $\textsc{msg}(\mathit{payload}, \mathit{sn}, i)$
    \item[HB bien-fondé :] $p_k$ a une première livraison de $\textsc{msg}(\mathit{payload}, \mathit{sn}, i)$
    \item[Lignes 3 et 5 :] $p_k$ délivre $m$
    \end{description}
  }

  \obBlock<8>[anchor=north, y=-9mm]{Démonstration : Uniformité}{

    \begin{center}
      \structure{Si un processus délivre $m$, tous les corrects délivrent $m$}
    \end{center}
    \vspace{-2mm}
    \begin{description}[STA-Terminaison :]
    \item[Lignes 4 et 5 :] $p_i$ diffuse $\textsc{msg}(\mathit{payload}, \mathit{sn}, j)$
    \item[Fiabilité :] $p_j$ délivre $\textsc{msg}(\mathit{payload}, \mathit{sn}, j)$
    \item[HB bien-fondé :] $p_j$ a une première livraison de $\textsc{msg}(\mathit{payload}, \mathit{sn}, i)$
    \item[Lignes 3 et 5 :] $p_j$ délivre $m$
    \end{description}
  }

  \obAlertBlock<9>[anchor=north, y=-9mm]{Peut-on inverser les lignes 3 et 4 ?}{
    \alert{Uniformité :} Si un processus délivre $m$, tous les corrects délivrent $m$
  }

  \onBlock<10>[anchor=north, y=-9mm]{Spécification de Reliable Broadcast (\textsc{rb}-broadcast)}{
    \vspace{-2mm}
    \begin{description}[Terminaison :]
    \item[Validité :] Si $p_i$ délivre $m$ de $p_j$, $p_j$ a diffusé $m$
    \item[Intégrité :] Un processus $p_i$ délivre $m$ au plus une fois
    \item[Terminaison :] Tout appel de \Broadcast par un correct se termine
    \item[Complétion :] Si un correct diffuse $m$, tous les corrects délivrent $m$
    \item[\alert{Fiabilité} : ] Si un \alert{correct} délivre $m$, tous les corrects délivrent $m$
    \end{description}
  }
