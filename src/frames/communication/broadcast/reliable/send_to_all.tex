\SetKwMessage{MSG}{msg}
\SetKwBroadcast{STA}{sta}
\SetKwVariable{Payload}{payload}

\begin{frame}{Notion de diffusion de message}

  \on[text, top=-2mm]{
    Quand $p_i$ veut exécuter une commande, la \structure{diffuse} à tout le système
    \begin{algorithm}[H]
      \Interface{\textsc{broadcast}}{
        \lMethod{$\Broadcast(\Payload)$}{\tcp*[f]{Crée un message $m$ et le diffuse}}
        \lEvent{\Deliver $\Payload$ \From $p_j$}{\tcp*[f]{Livraison de $m$ de la part de $p_j$}}
        \Method({\tcp*[f]{Variante synchronisée}}){$\SBroadcast(\Payload)$}{
          \nl $\Broadcast(\Payload)$\\
          \nl \Wait local delivery of the corresponding message
        }
      }
    \end{algorithm}
  }

  \onBlock<2->[bottom=-3mm]{L'implémentation la plus simple : Send-To-All}{
    \vspace{-1mm}
    \begin{algorithm}[H]
      \Method{$\STA.\Broadcast(\Payload)$ \At $p_i$}{
        \nl \lFor{$j$ \From $1$ \To $n$}{
          \Send $\alert{\MSG(\Payload)}$ \To $p_j$
        }
      }
      \Upon{\Receive $\alert{\MSG(\Payload)}$ \From $p_i$ \At $p_j$}{
        \nl \Trigger \STA.\Deliver $\Payload$ \From $p_i$
      }
    \end{algorithm}

    \alert{Quelles propriétés sont vérifiées par \STA-broadcast ?}

    \uncover<3->{
      \begin{description}
      \item[Validité :] Si $p_i$ délivre $m$ de $p_j$, $p_j$ a diffusé $m$
      \item[Intégrité :] Un processus $p_i$ délivre $m$ au plus une fois
      \item[Terminaison :] Tout appel de \Broadcast par un correct se termine
      \item[Complétion :] Si un correct diffuse $m$, tous les corrects délivrent $m$
      \end{description}
    }
  }
  
\end{frame}
