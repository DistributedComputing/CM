\SetKwFunction{Insert}{insert}
\SetKwFunction{Read}{read}
\SetKwData{State}{state}
\SetKwMessage{INSERT}{insert}

\begin{frame}{Nécessité de la diffusion fiable de messages}

  \onBlock[top=-4mm]{Premier essai d'implémentation d'un ensemble partagé}{
    \begin{algorithm}[H]
      \LVariables{\At $p_i$}{
        $\State_i \leftarrow \emptyset$\tcp*[r]{Set of values for which $p_i$ received a message \alert{$\INSERT$}}
      }
      \Method{$\Read()$ \At $p_i$}{
        \nl \Return $\State_i$
      }
      \Method{$\Insert(v)$ \At $p_i$}{
        \nl \lFor{$j$ \From $1$ \To $n$}{
          \Send \alert{$\INSERT(v)$} \To $p_j$
        }
      }
      \Upon{\Receive \alert{$\INSERT(v)$} \From $p_j$ \At $p_i$}{
        \nl $\State_i \leftarrow \State_i \cup \{ v \}$
      }
    \end{algorithm}
  }
  
  \onAlertBlock<2>[bottom]{Problème -- crash pendant la boucle}{
    \centering
    \begin{tikzpicture}[y=10mm]
      \draw[process] (0,2) node[left]{$p_1$} node[replica below]{$\emptyset$} to (9,2) node[replica, below left] {$\{a\}$} ;
      \draw[crashed] (0,1) node[left]{$p_2$} node[replica]      {$\emptyset$} to (5,1) ;
      \draw[process] (0,0) node[left]{$p_3$} node[replica]      {$\emptyset$} to (9,0) node[replica, above left] {$\emptyset$} ;

      \begin{scope}
        \clip (0,0) rectangle (5,2);
        \node[alert, operation] (i1) at (3.5,1) {$\Insert(a)$\hspace{30mm}};
      \end{scope}
      \draw[alert, message]   ([xshift=0mm]i1.center) to             (4.5,2);
      \draw[alert, message]   ([xshift=6mm]i1.center) to[bend left]  ([xshift=12mm]i1.center);

      \node[structure, operation] (r1) at (7,2) {$\Read() \rightarrow \{a\}$};
      \node[example, operation]   (r2) at (7,0) {$\Read() \rightarrow \emptyset$};
    \end{tikzpicture}
  }

\end{frame}
