\begin{frame}[fragile]{L'instruction compare\_and\_swap}

  \begin{block}{Spécification}
    \vspace{3mm}
    \begin{lstlisting}[gobble=6]
      if (get() == expect) {
        set(update);
        return true;
      }
      return false;
    \end{lstlisting}
  \end{block}

  \begin{exampleblock}{Exemple : compteur non-bloquant}
    \begin{lstlisting}[gobble=6]
      class Counter {
        private AtomicInteger value = new AtomicInteger(0); 
        public int getAndIncrement() {
          for (;;) {
            int current = value.get();
            int next = current + 1;
            if (value.compareAndSet(current, next))
            return current;
          }
        }
      }
    \end{lstlisting}
  \end{exampleblock}
