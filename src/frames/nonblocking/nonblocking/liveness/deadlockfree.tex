\begin{frame}[fragile]{Compteur deadlock-free}

  \begin{block}{Définition -- Deadlock-freedom}
    \example{Si le nombre d'appels est fini} \alert{et qu'il n'y a pas de crash}, \\
    alors toutes les opérations se terminent.
    \begin{itemize}
    \item Progrès global bloquant
    \end{itemize}
  \end{block}

  \begin{exampleblock}{Exemple}
    \begin{lstlisting}[gobble=6]
      class DeadlockFreeCounter {
        int value 0;
        private AtomicBoolean taken = new AtomicBoolean(false);
        
        public void increment() {
          while(taken.getAndSet(true));
          value = value+1;
          taken.set(false);
        }
        
        public int get() {
          return value;
        }
      }
    \end{lstlisting}
  \end{exampleblock}

\end{frame}
