\begin{frame}[fragile]{Preuve de linéarisabilité}

  \begin{block}{Points de linéarisation}
    \begin{description}
    \item[increment() :] \lstinline{values.set()}  
    \item[get() :] Soit $x$ la valeur retournée 
      \begin{itemize}
      \item Au début de l'exécution, $\sum \text{\lstinline{values}} \le x$
      \item À la fin de l'exécution, $\sum \text{\lstinline{values}} \ge x$
      \item À un point de l'exécution, $\sum \text{\lstinline{values}} = x$
      \end{itemize}
    \end{description}
  \end{block}
