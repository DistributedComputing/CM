\begin{frame}[fragile]{Solution wait-free}

  \begin{block}{Les écritures \alert{doivent aider} les lectures concurrentes}
    \begin{itemize}
    \item Chaque écriture fait une lecture et la publie 
    \item \only<-2>{Chaque lecture fait deux \lstinline{collect}.}\only<3>{Chaque lecture fait des \lstinline{collect} jusqu'à ce que :}
      \begin{itemize}
      \item \only<-2>{Si la valeur est identique, la retourner}\only<3>{deux \lstinline{collect} retournent la même valeur} 
      \item \only<-2>{Sinon, retourner la lecture de l'écriture concurrente}\only<3>{un thread ait écrit deux fois en concurrence}
      \end{itemize}
    \item<2> \alert{Problème : pas linéarisable !}
    \end{itemize}
  \end{block}
