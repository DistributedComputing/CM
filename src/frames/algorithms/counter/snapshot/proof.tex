\begin{frame}[fragile]{Preuve de linéarisabilité}

  \begin{block}{Points de linéarisation}
    \begin{description}
    \item[increment() :] \lstinline{values.set()}  
    \item[get() :] Soit $x$ la valeur retournée 
      \begin{itemize}
      \item Au début du dernier \lstinline{collect}, $\sum \text{\lstinline{values}} \le x$
      \item À la fin du \lstinline{collect} précédent, $\sum \text{\lstinline{values}} \ge x$
      \item Entre les deux derniers \lstinline{collect}, $\sum \text{\lstinline{values}} = x$
      \end{itemize}
    \end{description}
  \end{block}
