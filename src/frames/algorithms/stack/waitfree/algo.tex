    public class WaitFreeStack<T> {
      private final (*\Example<1>{AtomicInteger range}*) = new AtomicInteger(0);
      private final (*\Structure<1>{AtomicReferenceArray<T> items;}*) // Tableau infini (voir TD)

      public void push(T value) {
        int i = (*\Alert<2-3>{range.getAndIncrement();}*) // Attention : deux instructions
        (*\Alert<2-3>{items.set(i, value);}*)
      }

      public T pop() {
        (*\Structure<4-5>{for(int i = range.get()-1; i>=0; i--)}*) {
          T value = (*\Alert<4-5>{items.getAndSet(i, null);}*)
          if(value != null) return value;
        }
        return null;
      }
    }
  \end{lstlisting}
