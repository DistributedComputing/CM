    void push(T* value) {
      Node* node = new Node(value, head.load());
      while (!head.compare_exchange_weak(node->next, node));
    }

    T* pop() {
      for(;;) {
        Node* old = head.load();
        if(!old) return nullptr;
        if(head.compare_exchange_weak(old, old->next)) {
          T* value = old->value;
          delete old;
          return value;
        }
      }
    }
  \end{lstlisting}
