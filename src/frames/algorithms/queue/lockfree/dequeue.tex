    public T dequeue_sequentiel() {
      if(first == last) return null;
      first = first.next;
      return first.value;
    }

  \begin{lstlisting}[gobble=4]
    public T dequeue() {
      while (true) {
        var f = first.get();
        var l = last.get();
        var n = f.next.get();
        if (f == l) {
          if (n == null) return null;
          else last.compareAndSet(l, n);
        } else if (first.compareAndSet(f, n)) {
          return n.value;
        }
      }
    }
  \end{lstlisting}
