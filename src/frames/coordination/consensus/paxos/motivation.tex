\SetKwFunction{Propose}{propose}
\newcommand{\Abort}{\scriptsize\textsc{Abort}}
\newcommand{\Adopt}{\scriptsize\textsc{Adopt}}
\newcommand{\Commit}{\scriptsize\textsc{Commit}}
\SetKwVariable{Val}{val}


\begin{frame}{Consensus basé sur un leader}

  \onAlertBlock[top=-5mm]{Rappel -- Il est impossible d'avoir à la fois :}{
    \begin{description}
    \item[Accord: ] \vspace{-2mm}Au plus une valeur décidée
    \item[Validité: ] Toute valeur décidée a été proposée
    \item[Terminaison: ] \alert{\sout{{\color{black} Tout processus correct finit par décider une valeur}}}
    \end{description}
  }

  \onBlock[y=3mm, left=.74\textwidth]{Vers un algorithme toujours sûr}{
    \vspace{-1mm}
    \begin{algorithm}[H]
      \Operation{$\Propose(\Val_i)$ \At $p_i$}{
        \While(\tcp*[f]{C\oe ur algorithmique}){$\exists j, k, \Val_j \neq \Val_k$}{
          $\Val_i \leftarrow \text{choix d'un } \Val_j$\tcp*[f]{Stratégie de vivacité}
        }
        \Return $\Val_i$
      }
    \end{algorithm}
  }

  \onExampleBlock[bottom=4mm]{Exemples de stratégie}{
    \begin{description}[Ben-Or :]
    \item[Ben-Or :] \vspace{-2mm}Choisir une valeur \example{aléatoire}
      \begin{itemize}
      \item Tout processus correct décide \example{avec probabilité $1$}
      \end{itemize}
    \item[Paxos :] Suivre le choix d'un \example{leader} 
      \begin{itemize}
      \item Les processus décident \example{si le système est assez stable}
      \end{itemize}
    \end{description}
  }

  \on[x=42mm, y=-4mm]{
     \begin{tikzpicture}[automaton]\footnotesize
      \state[structure, rectangle, initial] (a) at (0, 1) {Vérifier l'accord};
      \state[structure, rectangle]                (b) at (0, 2) {Choisir une valeur};
      \state[structure, rectangle, accepting]     (c) at (0, 0) {Décider};
      \path (b) edge[bend right] (a);
      \path (a) edge[bend right]              node[swap]{\Abort}  (b);
      \path (a) edge[loop right, looseness=3] node      {\Adopt}  (a);
      \path (a) edge                          node      {\Commit} (c);
    \end{tikzpicture}
  }

  \footnoteref{M. Ben-Or. \textit{Another advantage of free choice for completely asynchronous agreement protocols.} PODC (1983)}
  \footnoteref{L. Lamport. \textit{The part-time parliament.} ACM TOCS (1998)}
  
\end{frame}
