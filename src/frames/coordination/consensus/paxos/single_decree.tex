\begin{frame}[fragile]{L'algorithme Single-decree Paxos}

  \on[top=-4mm, x=4mm, width=1.08\textwidth]{\small
    \begin{algorithm}[H]
      \LVariables{\At $p_i$}{
        \Exampleb<2,5>{$\Val_i \leftarrow \bot$}\tcp*[r]{Valeur courante supportée par $p_i$, initialement celle qu'il a proposée}
        \Alertb<2>{$\Ballot_i \leftarrow \langle 0, 0 \rangle$}\tcp*[r]{Plus grande estampille pour laquelle une ronde a été commencée}
        \Structureb<1,5>{$\Decided_i \leftarrow \bot$}\tcp*[f]{$\bot$ tant qu'aucune valeur n'a été décidée, puis la valeur décidée}
      }
      \Operation{$\Propose(v)$ \At $p_i$}{
        \nl\lIf{$\Val_i = \bot$}{\Exampleb<2,5>{$\Val_i \leftarrow v$}}
        \nl\While{\True}{
          \nl\Wait $\Alertb<1,4>{\Leader() = i} \lor \Structureb<1,5>{\Decided_i \neq \bot}$\\
          \nl\lIf{\Structureb<5>{$\Decided_i \neq \bot$}}{
            \Structureb<1,5>{\Return $\Decided_i$}
          }
          \nl\Alertb<2>{\Let $t$ \St $\langle t, i\rangle >  \Ballot_i$}\\
          \nl\MB.\SBroadcast $\alert{\BeginM(t)}$\\
          \nl\lIf{\Alertb<2>{$\langle t, i\rangle \neq \Ballot_i$}}{ 
            \Continue
          }
          \nl\MB.\SBroadcast $\example{\Voted(t, \Exampleb<5>{\Val_i})}$\\
          \nl\lIf{\Alertb<2>{$\langle t, i\rangle \neq \Ballot_i$}}{
            \Continue
          }
          \nl\RB.\Broadcast \Structureb<1>{$\structure{\Decide(\Exampleb<2,5>{\Val_i})}$}\\
          \vspace{-1mm}
        }
        \vspace{-1mm}
      }
    \end{algorithm}
  }

  \on[top=10mm, x=60mm]{\small
    \begin{algorithm}[H]
      \Upon{\MB.\Deliver $\alert{\BeginM(t)}$ \From $p_i$ \At $p_j$}{
        \If{\Alertb<2>{$\langle t, i\rangle \ge \Ballot_j$}}{
          \Alertb<2>{$\Ballot_j \leftarrow \langle t, i \rangle$}
        }
      }
      \Upon{\MB.\Deliver $\example{\Voted(t,v)}$ \From $p_i$ \At $p_j$}{
        \If{\Alertb<2>{$\langle t, i\rangle \ge \Ballot_j$}}{
          \Alertb<2>{$\Ballot_j \leftarrow \langle t, i \rangle$}\\
          \Exampleb<2,5>{$\Val_j \leftarrow v$}
        }
      }
      \Upon{\RB.\Deliver $\structure{\Decide(v)}$ \From $p_i$ \At $p_j$}{
        \Structureb<1,5>{$\Decided_j \leftarrow v$}
      }
    \end{algorithm}
  }

  \obAlertBlock<2>[anchor=north, y=-10mm]{Système de rondes asynchrones}{
    \begin{description}[Ballot number :]
    \item[Ballot number :]\vspace{-2mm} Ordre lexicographique total sur des estampilles de Lamport
    \item[Leader :]\vspace{-.5mm} Seul $p_i$ peut être \emph{leader} pour la ronde $\langle t, i \rangle$
    \item[Trois étapes d'une ronde :]
    \begin{description}[\Decide :]
    \item[\alert{\BeginM :}]     \vspace{-.5mm} Arrête les rondes précédentes
    \item[\example{\Voted :}]   \vspace{-.5mm} Transmet la valeur aux rondes suivantes
    \item[\structure{\Decide :}]\vspace{-.5mm} Partage la valeur décidée
    \end{description}

    \end{description}
  }

  \obAlertBlock<4>[anchor=north, y=-10mm]{Terminaison}{
    \begin{itemize}
    \item Après stabilisation de $\Omega$, un seul leader correct est élu
    \item Le leader finit par choisir un \emph{ballot number} supérieur à tous les autres
    \item Le leader termine sa ronde et envoie $\structure{\Decide(\Val_i)}$
    \end{itemize}
  }

  \obAlertBlock<5>[anchor=north, y=-10mm]{Validité}{
    \begin{itemize}
    \item Seules $\bot$ et des valeurs proposées sont écrites dans \Structure{$\Decided_i$} et \Example{$\Val_i$}
    \item Seules des valeurs proposées sont envoyées par $\example{\Voted}$ et $\structure{\Decide}$
    \item Lors de la décision, $\Decided_i\neq\bot$
    \end{itemize}
  }

  \obAlertBlock<6>[anchor=north, y=-10mm]{Accord}{
    \begin{itemize}
    \item\vspace{-2mm} Supposons, par l'absurde, que deux valeurs $v_i\neq v_j$ sont décidées
    \item\vspace{-1mm} Soient $\langle t_i, i \rangle <_{lex} \langle t_j, j \rangle$ les deux plus petits \emph{ballot numbers} tels que :
      \begin{itemize}
      \item $p_i$ diffuse $\structure{\Decide(v_i)}$
      \item $p_j$ diffuse $\example{\Voted(t_j, v_j)}$
      \end{itemize}
    \item\vspace{-1mm} On obtient MP1 : contradiction
    \end{itemize}
  }
