\begin{frame}[fragile]{Élection de leader}

  \on[top=-3mm]{
    \begin{algorithm}[H]
      \lLVariables{\At $p_i$}{
        $\Lastheard_i \leftarrow [\Time(), ..., \Time()]$
      }
      \Method{$\Leader()$ \At $p_i$}{
        \For(\tcp*[f]{ou ordre fixé selon la latence moyenne}){$j$ \From $1$ \To $n$}{
          \lIf{$\Time() - \Lastheard_i[j] < 3 \Delta$}{
            \Return $j$
          }
        }
      }
      \Background{\At $p_i$}{
        \While{\True}{
          $\Sleep(\Delta)$;
          \STA.\Broadcast \alert{$\HB()$}
        }
      }
      \Upon{\STA.\Deliver \alert{$\HB()$} \From $p_i$ \At $p_j$}{
        $\Lastheard_j[i] \leftarrow \Time()$
      }
    \end{algorithm}
  }

  \obBlock<1>[anchor=north, y=-3mm]{Analyse en système synchrone}{
    \vspace{-2mm}
    \begin{itemize}
    \item $\Time()$ avance à la même vitesse pour tous les processus
    \item Tous les messages sont acheminés en au plus $\Delta$ unités de temps
    \end{itemize}
  }

  \obBlock<2>[anchor=north, y=-3mm]{Analyse en système asynchrone}{
    \vspace{-2mm}
    \begin{itemize}
    \item Si un processus tombe en panne, il finit par ne plus être élu
    \item Pas de garantie de stabilité
    \end{itemize}
  }

  \onBlock<3>[anchor=north, y=-3mm]{Cas pratique : Stabilité intermittente}{
    \begin{description}[Périodes instables :] 
    \item[Périodes stables :] latence faible, peu de crashs
      \begin{itemize}
      \item \structure{élection} d'un leader stable
      \end{itemize}
    \item[Périodes instables :] pics de latence, partitions, crashs
      \begin{itemize}
      \item \structure{désaccord} sur le leader
      \end{itemize}
    \end{description}
  }
