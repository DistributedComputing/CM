\SetKwFunction{Propose}{propose}
\SetKwData{Decided}{decided}
\SetKwBroadcast{TO}{to}
\SetKwMessage{Prop}{propose}
\SetKwVariable{Val}{val}

\begin{frame}{Rappel : le consensus}

  \on[text, top=-3mm]{
    \begin{algorithm}[H]
      \Interface{Consensus}{
        \lMethod{$\Propose(\Val \in \mathcal{V}) \in \mathcal{V}$}{\tcp*[f]{Propose $\Val$, décide $\Val'$}}
      }
    \end{algorithm}

    \vspace{-1mm}
    \begin{description}[Terminaison :]
    \item[\alert{Accord :}] Au plus une valeur décidée
    \item[\alert{Validité :}] Toute valeur décidée a été proposée
    \item[\alert{Terminaison :}] Tout processus correct finit par décider une valeur
    \end{description}
  }

  \onBlock<2->[y=-5mm]{De Total-Order Broadcast au consensus}{
    \begin{algorithm}[H]
      \lLVariables{\At $p_i$}{$\Decided_i \leftarrow \bot$}
      \Method{$\Propose(\Val \in \mathcal{V}) \in \mathcal{V}$ \At $p_i$}{
        \TO.\SBroadcast \alert{$\Prop(\Val)$}\\
        \Return $\Decided_i$
      }
      \Upon{\TO.\Deliver \alert{$\Prop(\Val)$} \From $p_i$ \At $p_j$}{
        \lIf{$\Decided_j = \bot$}{$\Decided_j \leftarrow \Val$}
      }
    \end{algorithm}
  }

  \on<2->[y=-5mm, right=.35\textwidth]{
    \begin{tikzpicture}[y=9mm, x=9mm]
      \draw[process] (0,1) node[left]{$p_1$} to (4,1);
      \draw[process] (0,0) node[left]{$p_2$} to (4,0);

      \node[alert, operation]     (p1) at (2,1) {$\Propose(a) \rightarrow b$};
      \node[structure, operation] (p2) at (2,0) {$\Propose(b) \rightarrow b$};

      \draw[alert, message]       (p1.west) to[bend left]  (p1.east);
      \draw[alert, message]       (p1.west) to[bend below] (3.75,0);
      \draw[structure, message]   (p2.west) to             (2,1);
      \draw[structure, message]   (p2.west) to[bend right] (p2.east);
    \end{tikzpicture}
  }

  \onAlertBlock<3->[y=-27mm]{Théorème (FLP85)}{
    Il est impossible de résoudre le consensus dans $\mathcal{CAMP}_{n, t}[t\ge 1]$
  }

  \only<3->{
    \footnoteref{M. Fischer, N. Lynch, M. Paterson. \textit{Impossibility of Distributed Consensus with One Faulty Process.} JACM (1985)}
  }
  \footnoteref{M. Pease, R. Shostak, L. Lamport. \textit{Reaching agreement in the presence of faults.} JACM (1980)}
  
\end{frame}
