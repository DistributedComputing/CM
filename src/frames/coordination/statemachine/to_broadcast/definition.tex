\SetKwBroadcast{TO}{to}
\SetKwBroadcast{RBTO}{rb-to}
\SetKwBroadcast{TOCB}{to-cb}

\begin{frame}{Total-Order Broadcast (Atomic Broadcast)}

  \onBlock[top=-4mm]{Propriétés de \TO-broadcast}{
    \begin{description}
    \item[Validité :]\vspace{-1mm} Si $p_i$ délivre $m$ de $p_j$, $p_j$ a diffusé $m$
    \item[Intégrité :] Un processus $p_i$ délivre $m$ au plus une fois
    \item[Terminaison :] Tout appel de \Broadcast par un correct se termine
    \item[Complétion :] Si un correct diffuse $m$, tous les corrects délivrent $m$
    \item[\alert{Ordonnancement total :}] Si un processus délivre $m$ avant $m'$, \\aucun processus ne délivre $m'$ avant $m$.
    \end{description}
  }

  \onBlock[bottom=5mm]{Remarques}{
    \begin{itemize}
    \item \vspace{-1mm} Total-Ordering $\Rightarrow$ Mutual-Ordering 
    \item \structure{Ajout de propriétés :} \RBTO-broadcast, \TOCB-broadcast
    \end{itemize}
  }

  \on[y=-13mm, x=10mm]{
