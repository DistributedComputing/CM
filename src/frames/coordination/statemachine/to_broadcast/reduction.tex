\begin{frame}{Du consensus à Total-Order Broadcast}

  \on[top=-4.5mm]{\small
    \begin{algorithm}[H]
      \lSVariables{}{
        \Alertb<-2>{$\Cons[1, 2, \ldots]$}\tcp*[f]{Séquence infinie d'objets consensus}
      }
      \LVariables{}{
        $\Next_i \leftarrow 0$\tcp*[r]{Compteur pour numéroter les messages de $p_i$}
        \Exampleb<-2>{$\Pending_i \leftarrow \varepsilon$}\tcp*[r]{File de messages à délivrer identifiés par $\langle \mathit{payload}, \mathit{sn}, \mathit{pid} \rangle$}
      }

      \Method{$\TO.\Broadcast(\Payload)$}{
        \nl \URB.\Broadcast \example{$\Init(\Payload,\Next_i)$}\\
        \nl $\Next_i \leftarrow \Next_i+1$
      }

      \When{\URB.\Deliver \example{$\Init(\Payload,\SN)$} \From $p_j$}{
        \nl \Exampleb<-2>{$\Pending_i\leftarrow \Pending_i\cdot \langle\Payload,\SN,j\rangle$}
      }
      
      \Task{}{
        \nl \For{\,\Alertb<-2>{$k=1,2,3,\ldots$}}{
          \nl \Wait $\Pending_i\neq\varepsilon$\\
          \nl \Let $\langle\Payload,\SN,j\rangle \leftarrow \Alertb<-2>{\Cons[k]}.\Propose(\Exampleb{\Pending_i[1]})$\\
          \nl \Wait $\langle\Payload,\SN,j\rangle \in\Pending_i$\\
          \nl $\Pending_i\leftarrow \Pending_i\setminus \{\langle\Payload,\SN,j\rangle\}$\\
          \nl \Trigger \Alertb<-2>{\TO.\Deliver $\Payload$ \From $p_j$}
        }
      }
    \end{algorithm}
  }
