  \onBlock[top=-4mm]{Retour sur l'implémentation d'AlmaZone}{
    \vspace{-1mm}
    \begin{algorithm}[H]
      \LVariables{\At $p_i$}{
        $\State_i \leftarrow \emptyset$\tcp*[r]{La réplique locale de l'état de la machine à états}
        $\Result_i \leftarrow \False$\tcp*[f]{Dernière valeur retournée à une commande de $p_i$}
      }
      \lMethod{$\Consulter()$ \At $p_i$}{
        \Return $\State_i$
      }
      \lMethod{$\Vendre(\Val)$ \At $p_i$}{
        ... \tcp*[f]{Code ignoré ici par simplicité}
      }
      \Method{$\Acheter(\Val)$ \At $p_i$}{
        \nl \URBMBCB.\SBroadcast $\alert{\ACHETER(\Val)}$\\
        \nl \Return $\Result_i$
      }
      \Upon{\URBMBCB.\Deliver $\alert{\ACHETER(\Val)}$ \From $p_i$ \At $p_j$}{
        \nl \lIf{$j=i$}{
          $\Result_j \leftarrow \Val \in \State_j$
        }
        \nl $\State_j \leftarrow \State_j \setminus \{\Val\}$
      }
    \end{algorithm}
  }
