\begin{frame}[fragile]{Lemme 2 : $A$ ne possède pas de configuration critique}

  \onExampleBlock[top=-3mm]{Démonstration}{
    \begin{itemize}
    \item Supposons (absurde), que $A$ possède une configuration critique $C$.
    \item Quelles sont les prochains pas de $T_0$ et $T_1$ ?
      \begin{enumerate}
      \item<2-> Des opérations sur des variables différentes ? {\color{exampleColor}Impossible !}
      \item<3-> Des lectures sur la même variables ? {\color{exampleColor}Impossible !}
      \item<4-> Une lecture et une écriture ? \only<5->{\color{exampleColor}Impossible !}
      \item<6-> Deux écritures ? {\color{exampleColor}Impossible !}
      \end{enumerate}
    \item<6-> Aucun cas n'est possible : {\color{exampleColor}absurde !}
    \end{itemize}
