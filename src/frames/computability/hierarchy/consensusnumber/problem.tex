\begin{frame}{Un nouveau problème}

  \onBlock[top=-2mm]{Comparaison des instructions spéciales}{
    Vaut-il mieux avoir \lstinline{getAndIncrement} ou \lstinline{compareAndSet} sur sa machine ? 
    \begin{enumerate}
    \item \lstinline{compareAndSet} permet de résoudre le consensus entre $3$ processus
    \item \lstinline{getAndIncrement} ne le permet pas
    \item donc \lstinline{getAndIncrement} ne permet pas d'implémenter \lstinline{compareAndSet}
    \end{enumerate}
  }
