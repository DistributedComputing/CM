\SetKwBroadcast{MB}{mb}

\begin{frame}{Définition de Mutual broadcast}

  \onBlock[top=-3mm]{Propriétés de \MB-broadcast}{
    \begin{description}
    \item[Validité :] Si $p_i$ délivre $m$ de $p_j$, $p_j$ a diffusé $m$
    \item[Intégrité :] Un processus $p_i$ délivre $m$ au plus une fois
    \item[Terminaison :] Tout appel de \Broadcast par un correct se termine
    \item[Complétion :] Si un correct diffuse $m$, tous les corrects délivrent $m$
    \item[\alert{Ordonnancement mutuel :}] Si $p_i$ diffuse $m_i$ et $p_j$ diffuse $m_j$, \\
      il est impossible d'avoir en même temps :
      \begin{itemize}
      \item $p_i$ délivre $m_i$ avant $m_j$, et
      \item $p_j$ délivre $m_j$ avant $m_i$
      \end{itemize}
    \end{description}
  }
