\begin{frame}[fragile]{Algorithme de la Boulangerie (passage de messages)}

  \on[text, top=-1mm]{Utilisation de \structure{FIFO Mutual Broadcast} (\FMB)}

  \on[top=3mm]{\small
    \begin{algorithm}[H]
      \LVariables{\At $p_i$}{
        $\Tickets_i[n] \leftarrow [\infty, ..., \infty]$\tcp*[l]{Priorité de chaque processus}
      }
      \Method{$\Lock()$ \At $p_i$}{
        \nl \FMB.\SBroadcast \Exampleb<2,5>{\example{$\Hello()$}}\\
        \nl \Let $t_i > \max \{\Tickets_i[j] \mid \Tickets_i[j] < \infty\}$\\
        \nl \FMB.\SBroadcast \Alert<3,5>{\alert{$\Ticket(t_i)$}}\\
        \nl \Wait $\Structureb<4>{\left\langle t_i,i \right\rangle = \min_j \langle \Tickets_i[j], j \rangle}$
      }
      \Method{$\Unlock()$ \At $p_i$}{
        \nl \FMB.\SBroadcast \structure{$\Bye()$}
      }
    \end{algorithm}
  }

  \on[text,top=10mm, width=.6\textwidth, x=.33\textwidth]{\small
    \begin{algorithm}[H]
      \Upon{\FMB.\Deliver \example{$\Hello()$} \From $p_i$ \At $p_j$}{
        \Example<2>{$\Tickets_j[i] \leftarrow 0$}
      }
      ~\\
      \Upon{\FMB.\Deliver \alert{$\Ticket(t)$} \From $p_i$ \At $p_j$}{
        \Alert<3>{$\Tickets_j[i] \leftarrow t$}
      }
      ~\\
      \Upon{\FMB.\Deliver \structure{$\Bye()$} \From $p_i$ \At $p_j$}{
        $\Tickets_j[i] \leftarrow \infty$
      }
    \end{algorithm}
  }
