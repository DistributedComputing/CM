\begin{frame}[fragile]{Algorithme de la Boulangerie (mémoire partagée)}

  \on[text,top]{
    \begin{algorithm}[H]
      \SVariables{}{
        \Structureb<2>{$\Tickets[n] \leftarrow [\infty, ..., \infty]$}\tcp*[r]{Priorité de chaque processus}
      }
      \Method{$\Lock()$}{
        \nl \Example<6>{\Structureb<3>{$\Tickets[i] \leftarrow 0$}}\\
        \nl \Structureb<4>{\Let $t_i > \Alert<6>{\max\{\Tickets[j] \mid \Tickets[j] < \infty\}}$}\\
        \nl \Alert<6>{\Structureb<4>{$\Tickets[i] \leftarrow t_i$}}\\
        \nl \Structureb<5>{\Wait $\left\langle t_i,i \right\rangle = \min_j \Example<6>{\left\langle \Tickets[j], j \right\rangle}$}
      }
      \Method{$\Unlock()$}{
        \nl \Structureb<2>{$\Tickets[i] \leftarrow \infty$}
      }
    \end{algorithm}
  }

  \onBlock<6>[bottom=2mm]{Exclusion mutuelle}{

    \vspace{1mm}
    Soient $p_i$ et $p_j$ en section critique, avec $\left\langle t_i,i \right\rangle < \left\langle t_j,j \right\rangle$

    \vspace{3mm}
    \begin{tikzpicture}[y=7mm, anchor=mid]
      \draw[->] (.5,1) node[left]{$p_i$} -- (7,1) ;

      \node[structure] at (7,.5) {RW1 !};
      
      \footnotesize

      \node[operation, example, fill=example!20] at (5,0) {4: $\Tickets[i] \rightarrow \infty$};

    \end{tikzpicture}
  }
