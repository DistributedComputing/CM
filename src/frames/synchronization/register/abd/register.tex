  \on[top=-3mm]{
    \begin{algorithm}[H]
      \lLVariables{\At $p_i$}{
        \Alert<1>{$\Value_i  \leftarrow \bot$};
        \Example<1>{$\Time_i \leftarrow \langle 0, 0 \rangle$}
      }
      \Method{$\Read()$ \At $p_i$}{
        \uncover<3->{\nl \MB.\SBroadcast \Alert<-4,7-8>{\alert{$\Synch()$}}\tcp*[f]{Barrière \Write $\rightarrow$ \Read}}\\
        \nl \Let $\Val \leftarrow \Alertb<1>{\Value_i}$\\
        \uncover<8->{\nl \MB.\SBroadcast \Example<8>{\example{$\WRITE(\Val, \Time_i)$}}\tcp*[f]{Barrière \Read $\rightarrow$ \Read}}\\
        \nl \Return $\Val$
      }
      \Method{$\Write(\Val)$ \At $p_i$}{
        \uncover<6->{\nl \MB.\SBroadcast \Alert<6>{\alert{$\Synch()$}}\tcp*[f]{Barrière \Write $\rightarrow$ \Write}}\\
        \nl \Let $t_i$ \St \Exampleb<1>{$\langle t_i,i \rangle > \Time_i$}\\
        \nl \MB.\SBroadcast \Example<2-6>{\example{$\WRITE(\Val, \langle t_i,i \rangle)$}}\tcp*[f]{Estampillage des écritures}
      }
      \Upon{\MB.\Deliver \example{$\WRITE(\Val, \TS)$} \From $p_i$ \At $p_j$}{
        \nl \lIf{\Exampleb<1>{$\Time_j <_{lex} \TS$}}{
            \Alertb<1>{$\Value_j \leftarrow \Val$};~ \Exampleb<1>{$\Time_j \leftarrow \TS$}
        }
      }
    \end{algorithm}
  }

  \obBlock<9>[y=-26mm]{Théorème -- L'algorithme est linéarisable}{\small
    \begin{itemize}
    \item\vspace{-2mm} Posons $\structure{\TS(o)}$ l'estampille $\structure{\langle t, w\rangle}$ portée par le message \WRITE, et $\structure{o_1\;\mathcal{R}\;o_2}$ si : 
    \begin{itemize}
    \item\vspace{-1mm} $o_1$ se termine avant le début de $o_2$,
    \item $\TS(o_1) <_{lex} \TS(o_2)$, ou
    \item $\TS(o_1) = \TS(o_2)$, $o_1$ est une écriture et $o_2$ est une lecture.
    \end{itemize}
    \item\vspace{-2mm} \alert{$\mathcal{R}$ est acyclique}, et peut être étendu en un ordre total de linéarisation
    \end{itemize}
  }

  \footnoteref{H. Attiya, A. Bar-Noy, D. Dolev. \textit{Sharing memory robustly in message-passing systems} JACM (1995)}
